LIN_description_file;

Nodes {
  // 10 ms base time is required by the resolved BODY_10ms schedule. The
  // original LIN Matrix 20 ms baseline is retained as BASE_20ms below.
  // Jitter is a 1 ms target, not a hardware measurement.
  Master: Gateway_ECU, 10 ms, 1 ms ;
  Slaves: Door_ECU, Light_ECU, Window_ECU ;
}

Signals {

  DoorOpen: 1, 0, Door_ECU, Gateway_ECU ;
  DoorLocked: 1, 0, Door_ECU, Gateway_ECU ;

  // Provisional diagnostic response-error signal for tool validation.
  DoorResponseError: 1, 0, Door_ECU, Gateway_ECU ;

  // Door_Command - provisional bit layout from the message definition workbook.
  LockCommand: 2, 0, Gateway_ECU, Door_ECU ;
  RequestState: 1, 0, Gateway_ECU, Door_ECU ;

  // Light_Command - provisional bit layout from the message definition workbook.
  HeadLamp: 1, 0, Gateway_ECU, Light_ECU ;

  WindowAlive: 4, 0, Window_ECU, Gateway_ECU ;
  // Provisional diagnostic response-error signal for tool validation.
  WindowResponseError: 1, 0, Window_ECU, Gateway_ECU ;
}

Schedule_tables {
  // ACTIVE CANDIDATE: message definition workbook Team Schedule, 10 ms slot
  // and 100 ms supercycle. The two extended delays absorb source Reserved/
  // Idle entries. Schedule selection is an application decision; this table
  // is the approved operational candidate.
  BODY_10ms {
    Door_Status delay 10 ms ;
    Window_Command delay 10 ms ;
    Light_Command delay 10 ms ;
    Window_Status delay 10 ms ;
    Light_Status delay 20 ms ;
    Door_Command delay 10 ms ;
    Light_Command delay 30 ms ;
  }

  // REFERENCE ONLY: LIN_Matrix.xlsx / LIN_Schedule 0, 20, 40, 60 ms entries
  // and 100 ms repeat. The final 40 ms delay preserves the next-cycle start
  // at 100 ms; this reference schedule does not satisfy Light_Command 50 ms.
  BASE_20ms {
    Door_Status delay 20 ms ;
    Light_Command delay 20 ms ;
    Light_Status delay 20 ms ;
    Door_Command delay 40 ms ;
  }
}
